#INPUT
Get list of numbers from user or generate random array

#PROCESS

def draw_bars(numbers, selected=()):
    DISPLAY clear screen
    n <- len(numbers)
    bar_width <- 1.0
    
    For i, number in enumerate(numbers) do
        if i in selected then
            color <- "Red"
        else
            color <- "Blue"
        endif
        DISPLAY bar with width bar_width and height number using color
    endfor
    WAIT 100 milliseconds


def bubble_sort_animated(numbers):
    n <- len(numbers)
    
    For sweep in range(n) do
        swapped <- False
        For pair in range(0, n - 1 - sweep) do
            draw_bars(numbers, selected=(pair, pair+1))
            
            if numbers[pair] > numbers[pair+1] then
                numbers[pair], numbers[pair+1] <- numbers[pair+1], numbers[pair]
                swapped <- True
            endif
            
            draw_bars(numbers, selected=(pair, pair+1))
        endfor
        
        if not swapped then
            break
        endif
    endfor
    draw_bars(numbers)


def insertion_sort_animated(numbers):
    n <- len(numbers)

    For i in range(1, n) do
        key <- numbers[i]
        j <- i - 1
        
        draw_bars(numbers, selected=(i,))
        
        While j >= 0 and numbers[j] > key do
            draw_bars(numbers, selected=(j, j+1))
            numbers[j + 1] <- numbers[j]
            j <- j - 1
            draw_bars(numbers, selected=(j+1, j+2))
        endwhile
        
        numbers[j + 1] <- key
        draw_bars(numbers, selected=(j+1,))
    endfor
    draw_bars(numbers)


def selection_sort_animated(numbers):
    n <- len(numbers)

    For i in range(n - 1) do
        min_index <- i
        For j in range(i + 1, n) do
            draw_bars(numbers, selected=(min_index, j))
            
            if numbers[j] < numbers[min_index] then
                min_index <- j
            endif
        endfor
        
        if min_index != i then
            draw_bars(numbers, selected=(i, min_index))
            numbers[i], numbers[min_index] <- numbers[min_index], numbers[i]
            draw_bars(numbers, selected=(i, min_index))
        endif
    endfor
    draw_bars(numbers)


#OUTPUT
#DISPLAY
arr3 <- [random numbers from 0 to 100]
print(f"Before selection sort: {arr3}")

CALL selection_sort_animated(arr3)

print(f"After selection sort: {arr3}")
